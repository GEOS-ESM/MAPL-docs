   subroutine destroy_bundles(bundles, rc)
      type(ESMF_FieldBundle), intent(inout) :: bundles(:)
      integer, optional, intent(out) :: rc
      integer :: status, i
      
      do i=1, size(bundles)
         call destroy_bundle(bundles(i), _RC)
      end do
      _RETURN(_SUCCESS)
