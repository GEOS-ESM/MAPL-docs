   subroutine test_roundtrip_field(this)
      ! Test roundtrip: aspect -> field -> aspect
      class(ESMF_TestMethod), intent(inout) :: this
      type(UnitsAspect) :: aspect1, aspect2
      type(ESMF_Field) :: field
      character(len=:), allocatable :: retrieved_units
      integer :: status

      aspect1 = UnitsAspect(UNITS2)
      aspect2 = UnitsAspect(UNITS1)
      field = create_test_field()
      call aspect1%update_payload(field=field, _RC)
      call aspect2%update_from_payload(field=field, _RC)
      retrieved_units = aspect2%get_units(_RC)
      @assertEqual(UNITS2, retrieved_units)
      call ESMF_FieldDestroy(field, noGarbage=.true., _RC)
      _UNUSED_DUMMY(this)

   end subroutine test_roundtrip_field
