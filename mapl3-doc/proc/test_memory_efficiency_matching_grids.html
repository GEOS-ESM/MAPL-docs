   subroutine test_memory_efficiency_matching_grids(this)
      use mapl3g_StateRegistry
      use mapl3g_VariableSpec, only: VariableSpec, make_VariableSpec
      use mapl3g_StateItemSpec
      use mapl3g_VirtualConnectionPt
      use mapl3g_ConnectionPt
      use mapl3g_SimpleConnection
      use mapl3g_ExtensionFamily
      
      class(ESMF_TestMethod), intent(inout) :: this
      type(StateRegistry), target :: parent_registry, src_registry, dst_registry
      type(VirtualConnectionPt) :: export_pt, import_pt
      type(StateItemSpec) :: export_spec, import_spec
      type(VariableSpec) :: var_spec
      type(SimpleConnection) :: connection
      type(ExtensionFamily), pointer :: family
      integer :: num_extensions, status
      
      _UNUSED_DUMMY(this)
      
      ! Create registries
      parent_registry = StateRegistry('parent')
      src_registry = StateRegistry('source')
      dst_registry = StateRegistry('destination')
      call parent_registry%add_subregistry(src_registry)
      call parent_registry%add_subregistry(dst_registry)
      
      ! Create connection points
      export_pt = VirtualConnectionPt(state_intent='export', short_name='Q')
      import_pt = VirtualConnectionPt(state_intent='import', short_name='Q')
      
      ! Create export spec with geom_coarse
      var_spec = make_VariableSpec(ESMF_STATEINTENT_EXPORT, 'Q', &
           typekind=ESMF_TYPEKIND_R4, units='kg kg-1', _RC)
      export_spec = var_spec%make_StateItemSpec(src_registry, &
           component_geom=geom_coarse, vertical_grid=vertical_grid, _RC)
      call export_spec%create(_RC)
      
      ! Create import spec with SAME grid (geom_coarse)
      var_spec = make_VariableSpec(ESMF_STATEINTENT_IMPORT, 'Q', &
           typekind=ESMF_TYPEKIND_R4, units='kg kg-1', _RC)
      import_spec = var_spec%make_StateItemSpec(dst_registry, &
           component_geom=geom_coarse, vertical_grid=vertical_grid, _RC)
      call import_spec%create(_RC)
      
      ! Add specs to registries
      call src_registry%add_primary_spec(export_pt, export_spec, _RC)
      call dst_registry%add_primary_spec(import_pt, import_spec, _RC)
      
      ! Connect - with matching grids, should create NO extension fields
      connection = SimpleConnection( &
           ConnectionPt('source', export_pt), &
           ConnectionPt('destination', import_pt))
      call connection%connect(parent_registry, _RC)
      
      ! Get the export family and count extensions
      family => src_registry%get_extension_family(export_pt, _RC)
      num_extensions = family%num_variants()
      
      ! Key test: With matching grids, no extensions should be created
      ! num_variants() returns 1 for the primary spec + number of extensions
      ! So we expect num_variants() == 1 (just the primary, no extensions)
      @assertEqual(1, num_extensions, 'With matching grids, should have only primary spec (no extensions)')
      
   end subroutine test_memory_efficiency_matching_grids
