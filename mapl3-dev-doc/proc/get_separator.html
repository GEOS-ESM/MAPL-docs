   subroutine get_separator(this, separator, k)
      class(TextColumn), intent(in) :: this
      integer, intent(in) :: k
      character(*), intent(inout) :: separator(k)

      integer :: w
      character(1) :: c

      w = this%get_width()
      if (allocated(this%separator)) then
         c = this%separator
         separator(1) = repeat(c, w)
      end if

   end subroutine get_separator
