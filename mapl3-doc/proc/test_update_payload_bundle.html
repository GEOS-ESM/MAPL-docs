   subroutine test_update_payload_bundle(this)
      ! Test updating ESMF_FieldBundle units from UnitsAspect
      class(ESMF_TestMethod), intent(inout) :: this
      type(UnitsAspect) :: aspect
      type(ESMF_FieldBundle) :: bundle
      character(len=:), allocatable :: retrieved_units
      integer :: status

      aspect = UnitsAspect(UNITS2)
      bundle = create_test_bundle(UNITS1)
      call aspect%update_payload(bundle=bundle, _RC)
      call MAPL_FieldBundleGet(bundle, units=retrieved_units, _RC)
      @assertEqual(UNITS2, retrieved_units)
      call ESMF_FieldBundleDestroy(bundle, noGarbage=.true., _RC)
      _UNUSED_DUMMY(this)

   end subroutine test_update_payload_bundle
