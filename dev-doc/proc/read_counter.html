   integer function read_counter(win) result(value)
      integer, intent(in) :: win

      integer :: ierror

      call MPI_Win_lock(MPI_LOCK_EXCLUSIVE, 0, 0, win, ierror)
      call MPI_Get(value, 1, MPI_INTEGER, 0, 0_MPI_ADDRESS_KIND, 1, MPI_INTEGER, win, ierror)
      call MPI_Win_unlock(0, win, ierror)
   end function read_counter
