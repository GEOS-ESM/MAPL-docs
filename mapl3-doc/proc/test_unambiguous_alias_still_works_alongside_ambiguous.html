   subroutine test_unambiguous_alias_still_works_alongside_ambiguous()
      type(FieldDictionary) :: fd
      character(:), allocatable :: standard_name
      integer :: status

      fd = FieldDictionary(stream = '{' // &
           'X_Y_Z: {canonical_units: m, long_name: "X Y Z", aliases: [x, unique_x]},' // &
           'A_B_C: {canonical_units: s, long_name: "A B C", aliases: [x, unique_a]} }')

      ! Ambiguous alias fails.
      standard_name = fd%get_standard_name('x', rc=status)
      @assertExceptionRaised('FieldDictionary: short name "x" is ambiguous (maps to multiple standard names); provide standard_name explicitly')
      @assert_that(status, is(not(equal_to(0))))

      ! Unique aliases still resolve correctly.
      standard_name = fd%get_standard_name('unique_x', rc=status)
      @assert_that(status, is(0))
      @assertEqual('X_Y_Z', standard_name)

      standard_name = fd%get_standard_name('unique_a', rc=status)
      @assert_that(status, is(0))
      @assertEqual('A_B_C', standard_name)

   end subroutine test_unambiguous_alias_still_works_alongside_ambiguous
