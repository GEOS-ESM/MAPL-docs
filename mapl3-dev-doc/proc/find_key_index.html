    function find_key_index(this, key) result(index)
        class(StringDictionary), intent(in) :: this
        character(len=*), intent(in) :: key
        integer :: index
        integer :: i

        index = 0
        do i = 1, this%keys%size()
            if (this%keys%of(i) == key) then
                index = i
                return
            end if
        end do
    end function find_key_index
