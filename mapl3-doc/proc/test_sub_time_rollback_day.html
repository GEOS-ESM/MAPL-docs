   subroutine test_sub_time_rollback_day()
      ! ref_datetime=YYYY-MM-DDT06:00:00, time=2020-03-15T03:00:00
      ! new_time initially 2020-03-15T06:00:00 > time, leftmost wildcard is day -> subtract 1 day
      ! expected: 2020-03-14T06:00:00
      type(ESMF_Time) :: time, new_time
      integer :: status

      call ESMF_TimeSet(time, yy=2020, mm=3, dd=15, h=3, m=0, s=0, _RC)
      new_time = sub_time_in_datetime(time, 'YYYY-MM-DDT06:00:00', _RC)

      call assert_time_components(new_time, 2020, 3, 14, 6, 0, 0)
   end subroutine test_sub_time_rollback_day
