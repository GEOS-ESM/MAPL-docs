   subroutine test_coordinate_direction_default()
      type(BasicVerticalGridSpec) :: spec
      type(BasicVerticalGrid) :: grid
      type(VerticalCoordinateDirection) :: direction
      
      spec = BasicVerticalGridSpec(num_levels=72)
      call grid%initialize(spec)
      
      direction = grid%get_coordinate_direction()
      @assertEqual("downward", direction%to_string(), 'Default coordinate direction should be downward')
   end subroutine test_coordinate_direction_default
