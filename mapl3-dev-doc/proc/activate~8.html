   subroutine activate(this, rc)
      class(ServiceClassAspect), intent(inout) :: this
      integer, optional, intent(out) :: rc

      integer :: status
      type(ESMF_Info) :: info
      logical :: is_created

      is_created = ESMF_FieldBundleIsCreated(this%payload, _RC)
      if (is_created) then
         call ESMF_InfoGetFromHost(this%payload, info, _RC)
         call MAPL_FieldBundleInfoSetInternal(info, allocation_status=MAPL_STATEITEM_ALLOCATION_ACTIVE, _RC)
      end if

      _RETURN(_SUCCESS)
   end subroutine activate
