   subroutine test_update_from_payload_bundle_mirror(this)
      ! Test updating UngriddedDimsAspect from an ESMF_FieldBundle with mirror ungridded dims
      ! Mirror is represented by NO ungridded_dims entry in ESMF
      class(ESMF_TestMethod), intent(inout) :: this
      type(UngriddedDimsAspect) :: aspect
      type(ESMF_FieldBundle) :: bundle
      integer :: status

      aspect = UngriddedDimsAspect(UngriddedDims([5]))
      @assertFalse(aspect%is_mirror())
      bundle = create_test_bundle()
      call aspect%update_from_payload(bundle=bundle, _RC)
      @assertTrue(aspect%is_mirror(), MIRRORED//'.TRUE..')
      call ESMF_FieldBundleDestroy(bundle, noGarbage=.true., _RC)
      _UNUSED_DUMMY(this)

   end subroutine test_update_from_payload_bundle_mirror
