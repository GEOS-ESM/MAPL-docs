   subroutine fill_bundle(bundle, rc)
      type(ESMF_FieldBundle), intent(out) :: bundle
      integer, intent(out), optional :: rc

      integer :: status
      type(ESMF_Grid) :: grid
      type(ESMF_Field) :: fields(2) 

      grid = ESMF_GridCreateNoPeriDim(maxIndex=[5,5], _RC)
      fields(1) = ESMF_FieldCreate(grid, ESMF_TYPEKIND_R4, _RC)
      fields(2) = ESMF_FieldCreate(grid, ESMF_TYPEKIND_R4, _RC)
      call ESMF_FieldFill(fields(1), dataFillScheme="const", const1=1.d0, _RC)
      call ESMF_FieldFill(fields(2), dataFillScheme="const", const1=2.d0, _RC)
      bundle = ESMF_FieldBundleCreate(fieldList=fields, _RC)
      if (present(rc)) rc=0

  end subroutine fill_bundle
