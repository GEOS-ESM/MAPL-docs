   subroutine test_roundtrip_field_mirror(this)
      ! Test that aspect -> field -> aspect preserves MIRROR typekind
      class(ESMF_TestMethod), intent(inout) :: this
      type(TypekindAspect) :: aspect1, aspect2
      type(ESMF_Field) :: field
      type(ESMF_Typekind_Flag) :: tk1, tk2
      integer :: status

      aspect1 = TypekindAspect(MAPL_TYPEKIND_MIRROR)
      @assertTrue(aspect1%is_mirror())
      
      field = create_test_field(ESMF_TYPEKIND_R4)
      call aspect1%update_payload(field=field, _RC)
      
      aspect2 = TypekindAspect()
      call aspect2%update_from_payload(field=field, _RC)
      
      tk1 = aspect1%get_typekind()
      tk2 = aspect2%get_typekind()
      @assertEqual(tk2%dkind, tk1%dkind)
      @assertEqual(tk2%dkind, MAPL_TYPEKIND_MIRROR%dkind)
      @assertTrue(aspect2%is_mirror())
      
      call ESMF_FieldDestroy(field, noGarbage=.true., _RC)
   end subroutine test_roundtrip_field_mirror
