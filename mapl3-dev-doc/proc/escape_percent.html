   function escape_percent(str) result(escaped)
      character(len=:), allocatable :: escaped
      character(len=*), intent(in) :: str
      integer :: i

      escaped = ''
      do i = 1, len(str)
         escaped = escaped // str(i:i)
         if (str(i:i) == '%') escaped = escaped // '%'
      end do
   end function escape_percent
