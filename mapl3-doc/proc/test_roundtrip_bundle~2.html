   subroutine test_roundtrip_bundle(this)
      ! Test roundtrip: aspect -> bundle -> aspect
      class(ESMF_TestMethod), intent(inout) :: this
      type(UngriddedDimsAspect) :: aspect1, aspect2
      type(ESMF_FieldBundle) :: bundle
      type(UngriddedDims) :: original_dims, retrieved_dims
      integer :: status

      original_dims = UngriddedDims([72, 5])
      aspect1 = UngriddedDimsAspect(original_dims)
      bundle = create_test_bundle(UngriddedDims())
      call aspect1%update_payload(bundle=bundle, _RC)
      aspect2 = UngriddedDimsAspect()
      call aspect2%update_from_payload(bundle=bundle, _RC)
      @assertFalse(aspect2%is_mirror(), MIRRORED//'.FALSE..')
      retrieved_dims = aspect2%get_ungridded_dims(_RC)
      @assertTrue(original_dims == retrieved_dims, MISMATCH)
      call ESMF_FieldBundleDestroy(bundle, noGarbage=.true., _RC)
      _UNUSED_DUMMY(this)

   end subroutine test_roundtrip_bundle
