   subroutine test_mutual_exclusion(this)
      class (MpiTestMethod), intent(inout) :: this

      integer, parameter :: NITER = 20
      type (MpiMutex) :: mutex
      integer :: comm, npes, i, status
      integer :: win
      integer, target :: counter

      comm = this%getMpiCommunicator()
      npes = this%getNumProcesses()

      call create_counter(comm, counter, win)

      mutex = MpiMutex(comm, rc=status)
      @assertEqual(0, status)

      do i = 1, NITER
         call mutex%acquire(rc=status)
         @assertEqual(0, status)
         call increment_counter(win)
         call mutex%release(rc=status)
         @assertEqual(0, status)
      end do

      call MPI_Barrier(comm, status)
      @assertEqual(0, status)

      if (this%getProcessRank() == 0) then
         @assertEqual(npes*NITER, read_counter(win))
      end if

      call mutex%free_mpi_resources(rc=status)
      @assertEqual(0, status)
      call MPI_Win_free(win, status)

   end subroutine test_mutual_exclusion
