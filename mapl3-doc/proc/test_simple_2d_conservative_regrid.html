   subroutine test_simple_2d_conservative_regrid()
      type(EsmfRegridderParam) :: regrid_param
      type(RoutehandleParam) :: rh_param
      
      ! Verify we can create conservative regridder parameters
      rh_param = RoutehandleParam(regridmethod=ESMF_REGRIDMETHOD_CONSERVE)
      regrid_param = EsmfRegridderParam(rh_param)
      
      ! Verify is_conservative() works
      @assertTrue(regrid_param%is_conservative())
      
      ! For comparison, bilinear should not be conservative
      rh_param = RoutehandleParam(regridmethod=ESMF_REGRIDMETHOD_BILINEAR)
      regrid_param = EsmfRegridderParam(rh_param)
      @assertFalse(regrid_param%is_conservative())
      
   end subroutine test_simple_2d_conservative_regrid
