   subroutine test_comms_allreduce_sum_r4_1d(this)
      class (ESMF_TestMethod), intent(inout) :: this

      type (ESMF_VM) :: vm
      real, allocatable :: sendbuf(:), recvbuf(:)
      integer :: n_elements, petCount
      integer :: status

      vm = this%getVM()
      call ESMF_VMGet(vm, petCount=petCount)

      n_elements = 5
      allocate(sendbuf(n_elements))
      allocate(recvbuf(n_elements))

      ! Each rank sends array of ones
      sendbuf = 1.0
      recvbuf = -999.0

      ! Sum all values
      call comms_allreduce_sum(vm, sendbuf=sendbuf, recvbuf=recvbuf, cnt=n_elements, rc=status)

      ! Each element should sum to petCount
      @assertEqual(real(petCount), recvbuf)

      deallocate(sendbuf, recvbuf)

   end subroutine test_comms_allreduce_sum_r4_1d
