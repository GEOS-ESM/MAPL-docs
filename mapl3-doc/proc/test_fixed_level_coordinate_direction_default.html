   subroutine test_fixed_level_coordinate_direction_default()
      type(FixedLevelsVerticalGridSpec) :: spec
      type(FixedLevelsVerticalGrid) :: grid
      real, parameter :: test_levels(3) = [1000.0, 500.0, 100.0]
      type(VerticalCoordinateDirection) :: direction
      
      spec%levels = test_levels
      spec%physical_dimension = "pressure"
      spec%units = "Pa"
      
      call grid%initialize(spec)
      direction = grid%get_coordinate_direction()
      @assertEqual("downward", direction%to_string(), 'Default coordinate direction should be downward')
   end subroutine test_fixed_level_coordinate_direction_default
