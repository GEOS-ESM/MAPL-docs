   function add_vertex(this, v) result(iv)
      integer(kind=INT64) :: iv
      class(Mesh), target, intent(inout) :: this
      type(MeshVertex), intent(in) :: v

      call this%vertices%push_back(v)
      iv = this%vertices%size()
   end function add_vertex
