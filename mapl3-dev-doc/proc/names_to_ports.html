   function names_to_ports(names) result(ports)
      character(*), intent(in) :: names(:)
      type(PortSpecMap) :: ports

      integer :: i

      do i = 1, size(names)
         call ports%insert(trim(names(i)), PortSpec(trim(names(i))))
      end do
   end function names_to_ports
