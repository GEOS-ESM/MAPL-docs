   function create_test_vertical_grid(rc) result(vgrid)
      class(VerticalGrid), pointer :: vgrid
      integer, optional, intent(out) :: rc
      
      type(VerticalGridManager), pointer :: vgrid_manager
      type(BasicVerticalGridSpec) :: vspec
      integer :: status
      
      ! Create a simple basic vertical grid with NZ levels
      vspec = BasicVerticalGridSpec(num_levels=NZ)
      
      vgrid_manager => get_vertical_grid_manager(_RC)
      vgrid => vgrid_manager%create_grid(vspec, _RC)
      
      _RETURN(_SUCCESS)
   end function create_test_vertical_grid
