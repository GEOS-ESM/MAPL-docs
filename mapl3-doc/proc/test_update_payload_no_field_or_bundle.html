   subroutine test_update_payload_no_field_or_bundle(this)
      ! Test that update_payload returns early when neither field nor bundle is present
      class(ESMF_TestMethod), intent(inout) :: this
      type(TypekindAspect) :: aspect
      integer :: status

      aspect = TypekindAspect(ESMF_TYPEKIND_R8)
      
      ! Call with no field or bundle - should return successfully without error
      call aspect%update_payload(_RC)
      
      ! Should succeed without error (test passes if no error thrown)
   end subroutine test_update_payload_no_field_or_bundle
