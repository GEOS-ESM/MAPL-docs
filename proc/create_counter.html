   subroutine create_counter(comm, counter, win)
      integer, intent(in) :: comm
      integer, target, intent(inout) :: counter
      integer, intent(out) :: win

      integer :: sizeof_int, ierror
      integer(kind=MPI_ADDRESS_KIND) :: lb, extent

      call MPI_Type_get_extent(MPI_INTEGER, lb, extent, ierror)
      sizeof_int = int(extent)
      counter = 0
      call MPI_Win_create(counter, extent, sizeof_int, MPI_INFO_NULL, comm, win, ierror)
   end subroutine create_counter
