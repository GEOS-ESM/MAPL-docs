   subroutine test_update_bundle_no_vgrid(this)
      class(ESMF_TestMethod), intent(inout) :: this
      type(VerticalGridAspect) :: aspect
      type(ESMF_FieldBundle) :: bundle
      integer :: status
      class(VerticalGrid), pointer :: actual_vertical_grid
      character(len=*), parameter :: ASST_MSG = &
         & 'The FieldBundle vertical grid should not be associated.'

      actual_vertical_grid => null()
      bundle = ESMF_FieldBundleCreate(_RC)
      call MAPL_FieldBundleSet(bundle, vgrid=DestVertGrid,&
         & vert_staggerloc=VERTICAL_STAGGER_CENTER, vert_alignment=VALIGN_UP, _RC)
      aspect = VerticalGridAspect(regrid_method=VERTICAL_REGRID_CONSERVATIVE, &
         & vertical_stagger=VERTICAL_STAGGER_CENTER, vertical_alignment=VALIGN_UP)
      call aspect%update_payload(bundle=bundle, _RC)
      call MAPL_FieldBundleGet(bundle, vgrid=actual_vertical_grid, _RC)
      @assertFalse(associated(actual_vertical_grid), ASST_MSG)
      _UNUSED_DUMMY(this)

   end subroutine test_update_bundle_no_vgrid
