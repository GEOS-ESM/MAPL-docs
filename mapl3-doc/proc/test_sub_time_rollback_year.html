   subroutine test_sub_time_rollback_year()
      ! ref_datetime=YYYY-12-01T00:00:00, time=2000-10-15T21:00:00
      ! new_time initially 2000-12-01 > time, leftmost wildcard is year -> subtract 1 year
      ! expected: 1999-12-01T00:00:00
      type(ESMF_Time) :: time, new_time
      integer :: status

      call ESMF_TimeSet(time, yy=2000, mm=10, dd=15, h=21, m=0, s=0, _RC)
      new_time = sub_time_in_datetime(time, 'YYYY-12-01T00:00:00', _RC)

      call assert_time_components(new_time, 1999, 12, 1, 0, 0, 0)
   end subroutine test_sub_time_rollback_year
