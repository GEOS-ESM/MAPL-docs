   subroutine increment_counter(win)
      integer, intent(in) :: win

      integer :: value, ierror

      value = read_counter(win)
      value = value + 1

      call MPI_Win_lock(MPI_LOCK_EXCLUSIVE, 0, 0, win, ierror)
      call MPI_Put(value, 1, MPI_INTEGER, 0, 0_MPI_ADDRESS_KIND, 1, MPI_INTEGER, win, ierror)
      call MPI_Win_unlock(0, win, ierror)
   end subroutine increment_counter
