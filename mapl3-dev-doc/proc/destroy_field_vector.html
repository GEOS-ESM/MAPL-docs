   subroutine destroy_field_vector(fv, rc)
      type(ESMF_Field), intent(inout) :: fv(:)
      integer, optional, intent(out) :: rc

      integer :: i
      integer :: status

      do i = 1, size(fv)
         call ESMF_FieldDestroy(fv(i), noGarbage=.true., _RC)
      end do

      _RETURN(_SUCCESS)
   end subroutine destroy_field_vector
