   subroutine test_update_payload_field_mirror(this)
      ! Test updating an ESMF_Field from UngriddedDimsAspect with multiple dims
      class(ESMF_TestMethod), intent(inout) :: this
      type(UngriddedDimsAspect) :: aspect
      type(ESMF_Field) :: field
      type(UngriddedDims) :: retrieved_dims
      integer :: status

      aspect = UngriddedDimsAspect()
      @assertTrue(aspect%is_mirror(), 'Aspect should be mirror.')
      field = create_test_field(UngriddedDims([1]))
      call aspect%update_payload(field=field, _RC)
      call mapl_FieldGet(field, ungridded_dims=retrieved_dims, _RC)
      @assertTrue(retrieved_dims%is_mirror(), MIRRORED//'.TRUE..')
      call ESMF_FieldDestroy(field, noGarbage=.true., _RC)
      _UNUSED_DUMMY(this)

   end subroutine test_update_payload_field_mirror
