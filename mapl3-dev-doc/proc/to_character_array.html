   function to_character_array(s) result(ch)
      character, allocatable :: ch(:)
      character(len=*), intent(in) :: s
      integer :: i

      allocate(ch(len(s)))
      do i= 1, size(ch)
         ch(i) = s(i:i)
      end do

   end function to_character_array
