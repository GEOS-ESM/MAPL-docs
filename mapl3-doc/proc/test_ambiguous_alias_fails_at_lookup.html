   subroutine test_ambiguous_alias_fails_at_lookup()
      type(FieldDictionary) :: fd
      character(:), allocatable :: standard_name
      integer :: status

      ! Both X_Y_Z and A_B_C claim alias 'x' — this should warn but not abort.
      fd = FieldDictionary(stream = '{' // &
           'X_Y_Z: {canonical_units: m, long_name: "X Y Z", aliases: [x]},' // &
           'A_B_C: {canonical_units: s, long_name: "A B C", aliases: [x]} }')

      ! Lookup via the ambiguous alias must return a non-zero rc.
      standard_name = fd%get_standard_name('x', rc=status)
      @assertExceptionRaised('FieldDictionary: short name "x" is ambiguous (maps to multiple standard names); provide standard_name explicitly')
      @assert_that(status, is(not(equal_to(0))))

   end subroutine test_ambiguous_alias_fails_at_lookup
