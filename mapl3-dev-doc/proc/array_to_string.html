   function array_to_string(array) result(string)
      character(len=:), allocatable :: string
      character, intent(in) :: array(:)
      integer :: i

      allocate(character(len=size(array)) :: string)
      do i = 1, size(array)
         string(i:i) = array(i)
      end do

   end function array_to_string
