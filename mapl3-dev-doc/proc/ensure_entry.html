   subroutine ensure_entry(map, key)
      type(NodeId_NodeIdSet_Map), intent(inout) :: map
      type(NodeId), intent(in) :: key

      type(NodeIdSet) :: empty_set

      if (map%count(key) == 0) call map%insert(key, empty_set)
   end subroutine ensure_entry
