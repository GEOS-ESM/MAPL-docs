   subroutine test_update_from_payload_field_mirror(this)
      ! Test updating UngriddedDimsAspect from an ESMF_Field with mirror ungridded dims
      ! Mirror is represented by NO ungridded_dims entry in ESMF
      class(ESMF_TestMethod), intent(inout) :: this
      type(UngriddedDimsAspect) :: aspect
      type(ESMF_Field) :: field
      integer :: status

      aspect = UngriddedDimsAspect(UngriddedDims([1,2,3]))
      @assertFalse(aspect%is_mirror())
      field = create_test_field()
      call aspect%update_from_payload(field=field, _RC)
      @assertTrue(aspect%is_mirror(), MIRRORED//'.TRUE..')
      call ESMF_FieldDestroy(field, noGarbage=.true., _RC)
      _UNUSED_DUMMY(this)

   end subroutine test_update_from_payload_field_mirror
