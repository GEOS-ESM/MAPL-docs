   subroutine test_update_payload_field(this)
      ! Test updating ESMF_Field units from UnitsAspect
      class(ESMF_TestMethod), intent(inout) :: this
      type(UnitsAspect) :: aspect
      type(ESMF_Field) :: field
      character(len=:), allocatable :: retrieved_units
      integer :: status

      aspect = UnitsAspect(UNITS2)
      field = create_test_field(UNITS1)
      call aspect%update_payload(field=field, _RC)
      call MAPL_FieldGet(field, units=retrieved_units, _RC)
      @assertEqual(UNITS2, retrieved_units)
      call ESMF_FieldDestroy(field, noGarbage=.true., _RC)
      _UNUSED_DUMMY(this)

   end subroutine test_update_payload_field
