   function has_id(this, id) result(found)
      logical :: found
      class(VerticalGridManager), target, intent(in) :: this
      integer, intent(in) :: id

      type(IntegerVerticalGridMapIterator) :: iter

      iter = this%grid_map%find(id)
      found = (iter /= this%grid_map%end())
   end function has_id
