   subroutine test_update_payload_no_field_or_bundle(this)
      ! Test that update_payload does nothing when neither field nor bundle is present
      class(ESMF_TestMethod), intent(inout) :: this
      type(UngriddedDimsAspect) :: aspect
      type(UngriddedDims) :: test_dims
      integer :: status

      test_dims = UngriddedDims([72])
      aspect = UngriddedDimsAspect(test_dims)
      call aspect%update_payload(_RC)
      @assertTrue(.true.)
      _UNUSED_DUMMY(this)

   end subroutine test_update_payload_no_field_or_bundle
