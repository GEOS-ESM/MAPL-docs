   subroutine test_update_from_payload_no_field_or_bundle(this)
      ! Test that update_from_payload does nothing when neither field nor bundle is present
      class(ESMF_TestMethod), intent(inout) :: this
      type(UngriddedDimsAspect) :: aspect
      type(UngriddedDims) :: initial_dims, retrieved_dims
      integer :: status

      initial_dims = UngriddedDims([10])
      aspect = UngriddedDimsAspect(initial_dims)
      
      call aspect%update_from_payload(_RC)
      
      retrieved_dims = aspect%get_ungridded_dims(_RC)
      @assertEqual(1, retrieved_dims%get_num_ungridded())
   end subroutine test_update_from_payload_no_field_or_bundle
