   subroutine test_non_vector_cannot_have_basis_kind()
      type(VariableSpec) :: var_spec
      integer :: status

      ! Try to create a field (non-vector) with vector_basis_kind - should fail
      var_spec = make_VariableSpec( &
           state_intent=ESMF_STATEINTENT_EXPORT, &
           short_name='temp', &
           standard_name='air_temperature', &
           geom=geom, &
           units='K', &
           vector_basis_kind='NS', &
           rc=status)

      @assertExceptionRaised('vector_basis_kind can only be specified for vectors')

   end subroutine test_non_vector_cannot_have_basis_kind
