   logical function has_kind(kinds, kind)
      type(StateItemCharacteristicKind), intent(in) :: kinds(:)
      type(StateItemCharacteristicKind), intent(in) :: kind

      integer :: i

      has_kind = .false.
      do i = 1, size(kinds)
         if (kinds(i) == kind) has_kind = .true.
      end do
   end function has_kind
