   subroutine test_update_from_payload_field_empty(this)
      ! Test updating UnitsAspect from an ESMF_Field with empty units
      class(ESMF_TestMethod), intent(inout) :: this
      type(UnitsAspect) :: aspect
      type(ESMF_Field) :: field
      integer :: status

      aspect = UnitsAspect(UNITS1)
      field = create_test_field()
      call aspect%update_from_payload(field=field, _RC)
      @assertTrue(aspect%is_mirror(), 'aspect should be mirror.')
      call ESMF_FieldDestroy(field, noGarbage=.true., _RC)
      _UNUSED_DUMMY(this)

   end subroutine test_update_from_payload_field_empty
