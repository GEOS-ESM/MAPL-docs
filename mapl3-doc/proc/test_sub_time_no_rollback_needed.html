   subroutine test_sub_time_no_rollback_needed()
      ! ref_datetime=YYYY-12-01T00:00:00, time=2000-12-15T00:00:00
      ! new_time=2000-12-01 is before time, no decrement
      ! expected: 2000-12-01T00:00:00
      type(ESMF_Time) :: time, new_time
      integer :: status

      call ESMF_TimeSet(time, yy=2000, mm=12, dd=15, h=0, m=0, s=0, _RC)
      new_time = sub_time_in_datetime(time, 'YYYY-12-01T00:00:00', _RC)

      call assert_time_components(new_time, 2000, 12, 1, 0, 0, 0)
   end subroutine test_sub_time_no_rollback_needed
