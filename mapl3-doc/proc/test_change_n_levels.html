   subroutine test_change_n_levels(this)
      class(ESMF_TestMethod), intent(inout) :: this
      type(ESMF_Field) :: f
      type(ESMF_Grid) :: grid
      type(ESMF_Geom) :: geom

      integer :: status
      type(ESMF_FieldStatus_Flag) :: field_status
      type(ESMF_TypeKind_Flag) :: typekind
      real(ESMF_KIND_R4), pointer :: x(:,:,:,:)
      type(FieldDelta) :: delta
      integer, parameter :: NEW_NUM_LEVELS = 7
      type(VerticalGridManager), pointer :: vgrid_manager
      class(VerticalGrid), pointer :: vgrid, new_vgrid

      grid = ESMF_GridCreateNoPeriDim(maxIndex=[4,4], name='I_AM_GROOT', _RC)
      geom = ESMF_GeomCreate(grid, _RC)

      vgrid_manager => get_vertical_grid_manager()
      vgrid => vgrid_manager%create_grid(BasicVerticalGridSpec(num_levels=ORIG_VGRID_LEVELS), _RC)

      ! BasicVerticalGrid only supports CENTER staggering
      f = MAPL_FieldCreate(geom, typekind=ESMF_TYPEKIND_R4, name='in', &
           ungridded_dims=UngriddedDims([3]), &
           vgrid=vgrid, vert_staggerloc=VERTICAL_STAGGER_CENTER, _RC)

      ! Note: Since FieldDelta no longer tracks num_levels changes directly,
      ! vertical grid changes would be reflected through geom changes.
      ! For this test, we'll use FieldSet to update the vgrid
      new_vgrid => vgrid_manager%create_grid(BasicVerticalGridSpec(num_levels=NEW_NUM_LEVELS), _RC)
      call MAPL_FieldSet(f, vgrid=new_vgrid, _RC)
      
      call ESMF_FieldGet(f, status=field_status, typekind=typekind, _RC)
      @assert_that(field_status == ESMF_FIELDSTATUS_COMPLETE, is(true()))
      @assert_that(typekind == ESMF_TYPEKIND_R4, is(true()))

      call ESMF_FieldGet(f, fArrayPtr=x, _RC)
      @assert_that(shape(x), is(equal_to([4,4,NEW_NUM_LEVELS,3])))

      call ESMF_FieldDestroy(f, _RC)
      call ESMF_GridDestroy(grid, _RC)
      call ESMF_GeomDestroy(geom, _RC)

      _UNUSED_DUMMY(this)
   end subroutine test_change_n_levels
