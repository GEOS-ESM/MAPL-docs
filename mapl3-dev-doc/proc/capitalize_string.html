   function capitalize_string(s) result(t)
      character(len=:), allocatable :: t
      character(len=*), intent(in) :: s

      t = ''
      if(len(s) > 0) t = to_upper(s(1:1)) // to_lower(s(2:))

   end function capitalize_string
