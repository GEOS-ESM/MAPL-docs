   subroutine destroy(this, rc)
      class(TimeMin), intent(inout) :: this
      integer, optional, intent(out) :: rc

      integer :: status

      if (this%is_bundle) then
         call MAPL_FieldBundleDestroy(this%min_b, _RC)
      else
         call esmf_FieldDestroy(this%min_f, _RC)
      end if

      _RETURN(_SUCCESS)
