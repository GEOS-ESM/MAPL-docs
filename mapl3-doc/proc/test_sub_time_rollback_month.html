   subroutine test_sub_time_rollback_month()
      ! ref_datetime=YYYY-MM-20T00:00:00, time=2020-03-15T00:00:00
      ! new_time initially 2020-03-20 > time, leftmost wildcard is month -> subtract 1 month
      ! expected: 2020-02-20T00:00:00
      type(ESMF_Time) :: time, new_time
      integer :: status

      call ESMF_TimeSet(time, yy=2020, mm=3, dd=15, h=0, m=0, s=0, _RC)
      new_time = sub_time_in_datetime(time, 'YYYY-MM-20T00:00:00', _RC)

      call assert_time_components(new_time, 2020, 2, 20, 0, 0, 0)
   end subroutine test_sub_time_rollback_month
