   subroutine test_empty_units_roundtrip_bundle(this)
      ! Test that empty units work correctly in roundtrip
      class(ESMF_TestMethod), intent(inout) :: this
      type(UnitsAspect) :: aspect1, aspect2
      type(ESMF_FieldBundle) :: bundle
      integer :: status

      aspect1 = UnitsAspect()
      aspect2 = UnitsAspect(UNITS1)

      bundle = create_test_bundle()
      call aspect1%update_payload(bundle=bundle, _RC)
      call aspect2%update_from_payload(bundle=bundle, _RC)
      @assertTrue(aspect2%is_mirror(), 'aspect2 should be mirror.')
      call ESMF_FieldBundleDestroy(bundle, noGarbage=.true., _RC)
      _UNUSED_DUMMY(this)

   end subroutine test_empty_units_roundtrip_bundle
