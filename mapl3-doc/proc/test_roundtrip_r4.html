   subroutine test_roundtrip_R4(this)
      ! Test that normalize → denormalize recovers original value
      class(ESMF_TestMethod), intent(inout) :: this
      integer :: status
      type(ESMF_Field) :: field_in, field_mid_export, field_mid_import, field_out, field_aux
      type(ESMF_State) :: midState_export, midState_import
      real(kind=ESMF_KIND_R4), pointer :: fptr_in(:), fptr_mid_exp(:), fptr_mid_imp(:), fptr_out(:), fptr_aux(:)
      real(kind=ESMF_KIND_R4), parameter :: INPUT_VALUE = 1.0e-3_R4  ! kg/kg (original)
      real(kind=ESMF_KIND_R4), parameter :: DELP_VALUE = 1000.0_R4   ! Pa
      integer :: i
      type(NormalizationTransform) :: normalize_transform, denormalize_transform

      _UNUSED_DUMMY(this)
      
      call initialize_states(states, grid, [ESMF_TYPEKIND_R4, ESMF_TYPEKIND_R4],&
         & ESMF_NAMES, rc=status)
      @assertEqual(0, status, "Unable to initialize ESMF_State's")
      
      ! Create middle states for normalized value
      ! Export state for normalize transform (has COUPLER_EXPORT_NAME)
      midState_export = ESMF_StateCreate(name='midState_export', stateIntent=ESMF_STATEINTENT_EXPORT, _RC)
      field_mid_export = ESMF_FieldCreate(grid=grid, name=COUPLER_EXPORT_NAME, typekind=ESMF_TYPEKIND_R4, _RC)
      call ESMF_StateAdd(midState_export, [field_mid_export], _RC)
      ! Import state for denormalize transform (has COUPLER_IMPORT_NAME)
      midState_import = ESMF_StateCreate(name='midState_import', stateIntent=ESMF_STATEINTENT_IMPORT, _RC)
      field_mid_import = ESMF_FieldCreate(grid=grid, name=COUPLER_IMPORT_NAME, typekind=ESMF_TYPEKIND_R4, _RC)
      call ESMF_StateAdd(midState_import, [field_mid_import], _RC)
      
      ! Get fields and set input values
      call ESMF_StateGet(importState, itemName=COUPLER_IMPORT_NAME, field=field_in, _RC)
      call ESMF_StateGet(exportState, itemName=COUPLER_EXPORT_NAME, field=field_out, _RC)
      call assign_fptr(field_in, fptr_in, _RC)
      call assign_fptr(field_mid_export, fptr_mid_exp, _RC)
      call assign_fptr(field_mid_import, fptr_mid_imp, _RC)
      call assign_fptr(field_out, fptr_out, _RC)
      fptr_in = INPUT_VALUE
      fptr_mid_exp = 0.0
      fptr_mid_imp = 0.0
      fptr_out = 0.0
      
      ! Create auxiliary field (DELP)
      field_aux = ESMF_FieldCreate(grid=grid, name=AUX_FIELD_NAME, typekind=ESMF_TYPEKIND_R4, _RC)
      call assign_fptr(field_aux, fptr_aux, _RC)
      fptr_aux = DELP_VALUE
      
      ! Step 1: Normalize (forward)
      normalize_transform = new_NormalizationTransform(AUX_FIELD_NAME, SCALE_FACTOR, field_aux, .false.)
      call normalize_transform%initialize(importState, midState_export, clock, rc=status)
      @assertEqual(_SUCCESS, status, 'Failed to initialize normalize transform')
      call normalize_transform%update(importState, midState_export, clock, rc=status)
      @assertEqual(_SUCCESS, status, 'Failed to normalize')
      
      ! Copy normalized data from export field to import field
      ! (In production, a coupler would handle this)
      fptr_mid_imp = fptr_mid_exp
      
      ! Step 2: Denormalize (inverse)
      denormalize_transform = new_NormalizationTransform(AUX_FIELD_NAME, SCALE_FACTOR, field_aux, .true.)
      call denormalize_transform%initialize(midState_import, exportState, clock, rc=status)
      @assertEqual(_SUCCESS, status, 'Failed to initialize denormalize transform')
      call denormalize_transform%update(midState_import, exportState, clock, rc=status)
      @assertEqual(_SUCCESS, status, 'Failed to denormalize')
      
       ! Verify round-trip: output should match original input
       do i=1, size(fptr_out)
          @assertEqual(INPUT_VALUE, fptr_out(i), tolerance=1.0e-6_R4, message='Round-trip should recover original value')
       end do
      
       call ESMF_FieldDestroy(field_aux, _RC)
       call ESMF_FieldDestroy(field_mid_export, _RC)
       call ESMF_FieldDestroy(field_mid_import, _RC)
       call ESMF_StateDestroy(midState_export, _RC)
       call ESMF_StateDestroy(midState_import, _RC)

   end subroutine test_roundtrip_R4
