   subroutine test_new_inverse_normalization_aspect(this)
      ! Test creating an inverse normalization aspect using subclass
      class(ESMF_TestMethod), intent(inout) :: this
      type(ImportNormalization) :: aspect
      type(AspectId) :: aspect_id
      character(:), allocatable :: aux_field
      real :: scale
      integer :: status

      aspect = ImportNormalization(aux_field_name='DELP', &
                                          scale_factor=0.102, &
                                          source_units='kg/m2', &
                                          target_units='kg/kg')

      @assertFalse(aspect%is_mirror(), 'aspect should not be mirror when created with parameters')
      
      aux_field = aspect%get_aux_field_name(_RC)
      @assertTrue(aux_field == 'DELP', 'aux field should be DELP')
      
      scale = aspect%get_scale_factor(_RC)
      @assertTrue(abs(scale - 0.102) < 1.0e-6, 'scale should be 0.102')
      
      aspect_id = aspect%get_aspect_id()
      @assertTrue(aspect_id == IMPORT_NORMALIZATION_ASPECT_ID, 'aspect_id should be IMPORT_NORMALIZATION_ASPECT_ID')

      _UNUSED_DUMMY(this)

   end subroutine test_new_inverse_normalization_aspect
